```swift
import AppKit

// All app colors use the sampler's SDR sRGB model. Alpha stays unassociated.
struct SampledColor: Equatable, Hashable, Codable, Sendable {
    let red: UInt8
    let green: UInt8
    let blue: UInt8
    var alpha: Double = 1

    init(red: UInt8, green: UInt8, blue: UInt8, alpha: Double = 1) {
        self.red = red; self.green = green; self.blue = blue
        self.alpha = min(1, max(0, alpha.isFinite ? alpha : 1))
    }

    init?(hex: String) {
        var value = hex.trimmingCharacters(in: .whitespacesAndNewlines)
        if value.hasPrefix("#") { value.removeFirst() }
        guard [3, 4, 6, 8].contains(value.count), value.allSatisfy({ $0.isASCII && $0.isHexDigit }) else { return nil }
        if value.count <= 4 { value = value.map { "\($0)\($0)" }.joined() }
        guard let bits = UInt32(value, radix: 16) else { return nil }
        let hasAlpha = value.count == 8
        let rgb = hasAlpha ? bits >> 8 : bits
        self.init(red: UInt8((rgb >> 16) & 255), green: UInt8((rgb >> 8) & 255), blue: UInt8(rgb & 255),
                  alpha: hasAlpha ? Double(bits & 255) / 255 : 1)
    }

    var hex: String { String(format: "#%02X%02X%02X", red, green, blue) }
    var hexAlpha: String { hex + String(format: "%02X", Int((alpha * 255).rounded())) }
    var identity: String { hexAlpha }
    var components: [Double] { [Double(red) / 255, Double(green) / 255, Double(blue) / 255] }
    var nsColor: NSColor { NSColor(srgbRed: Double(red) / 255, green: Double(green) / 255, blue: Double(blue) / 255, alpha: alpha) }
    static let white = SampledColor(red: 255, green: 255, blue: 255)
    static let black = SampledColor(red: 0, green: 0, blue: 0)

    static func linear(_ x: Double) -> Double { x <= 0.04045 ? x / 12.92 : pow((x + 0.055) / 1.055, 2.4) }
    static func encoded(_ x: Double) -> Double { x <= 0.0031308 ? 12.92 * x : 1.055 * pow(x, 1 / 2.4) - 0.055 }
    var luminance: Double {
        let c = components.map(Self.linear)
        return 0.2126 * c[0] + 0.7152 * c[1] + 0.0722 * c[2]
    }

    // Encoded sRGB source-over, matching ordinary CSS/AppKit compositing.
    func composited(over background: SampledColor) -> SampledColor {
        let a = alpha + background.alpha * (1 - alpha)
        guard a > 0 else { return SampledColor(red: 0, green: 0, blue: 0, alpha: 0) }
        let values = zip(components, background.components).map { ($0 * alpha + $1 * background.alpha * (1 - alpha)) / a }
        return Self.from(values, alpha: a)
    }

    static func from(_ values: [Double], alpha: Double = 1) -> SampledColor {
        let c = values.map { UInt8(min(255, max(0, ($0 * 255).rounded()))) }
        return SampledColor(red: c[0], green: c[1], blue: c[2], alpha: alpha)
    }

    // Bjorn Ottosson's published linear-sRGB <-> OKLab transforms.
    var lab: [Double] {
        let c = components.map(Self.linear), r = c[0], g = c[1], b = c[2]
        let l = cbrt(0.4122214708*r + 0.5363325363*g + 0.0514459929*b)
        let m = cbrt(0.2119034982*r + 0.6806995451*g + 0.1073969566*b)
        let s = cbrt(0.0883024619*r + 0.2817188376*g + 0.6299787005*b)
        return [0.2104542553*l + 0.793617785*m - 0.0040720468*s,
                1.9779984951*l - 2.428592205*m + 0.4505937099*s,
                0.0259040371*l + 0.7827717662*m - 0.808675766*s]
    }

    static func fromLab(_ lab: [Double], alpha: Double = 1) -> SampledColor {
        let l = pow(lab[0] + 0.3963377774*lab[1] + 0.2158037573*lab[2], 3)
        let m = pow(lab[0] - 0.1055613458*lab[1] - 0.0638541728*lab[2], 3)
        let s = pow(lab[0] - 0.0894841775*lab[1] - 1.291485548*lab[2], 3)
        return from([4.0767416621*l - 3.3077115913*m + 0.2309699292*s,
                     -1.2684380046*l + 2.6097574011*m - 0.3413193965*s,
                     -0.0041960863*l - 0.7034186147*m + 1.707614701*s].map(encoded), alpha: alpha)
    }

    var hslHSV: (h: Double, s: Double, l: Double, v: Double, sv: Double) {
        let c = components, r = c[0], g = c[1], b = c[2]
        let hi = c.max()!, lo = c.min()!, delta = hi - lo, light = (hi + lo) / 2
        var hue = 0.0
        if delta > 0 {
            if hi == r { hue = ((g - b) / delta).truncatingRemainder(dividingBy: 6) }
            else if hi == g { hue = (b - r) / delta + 2 }
            else { hue = (r - g) / delta + 4 }
            hue = (hue * 60 + 360).truncatingRemainder(dividingBy: 360)
        }
        return (hue, delta == 0 ? 0 : delta / (1 - abs(2 * light - 1)), light, hi, hi == 0 ? 0 : delta / hi)
    }
}

    func string(for color: SampledColor, name: String = "prismdrop") -> String {
        func n(_ x: Double, _ digits: Int = 4) -> String {
            let rounded = abs(x) < pow(10, -Double(digits)) / 2 ? 0 : x
            var value = String(format: "%.*f", locale: Locale(identifier: "en_US_POSIX"), digits, rounded)
            while value.contains(".") && value.hasSuffix("0") { value.removeLast() }
            if value.hasSuffix(".") { value.removeLast() }
            return value
        }
        let r = color.red, g = color.green, b = color.blue, a = n(color.alpha)
        let h = color.hslHSV
        let rgbValue = color.alpha < 1 ? "rgb(\(r) \(g) \(b) / \(a))" : "rgb(\(r) \(g) \(b))"
        switch self {
        case .hex: return color.hex
        case .hexAlpha: return color.hexAlpha
        case .rgb: return rgbValue
        case .rgba: return "rgba(\(r), \(g), \(b), \(a))"
        case .hsl: return "hsl(\(n(h.h, 2)) \(n(h.s * 100, 2))% \(n(h.l * 100, 2))%\(color.alpha < 1 ? " / \(a)" : ""))"
        case .hsla: return "hsla(\(n(h.h, 2)), \(n(h.s * 100, 2))%, \(n(h.l * 100, 2))%, \(a))"
        case .hsv, .hsb:
            return "\(rawValue)(\(n(h.h, 2))°, \(n(h.sv * 100, 2))%, \(n(h.v * 100, 2))%\(color.alpha < 1 ? ", \(a)" : ""))"
        case .oklch:
            let lab = color.lab, chroma = hypot(lab[1], lab[2])
            let hue = chroma < 0.00001 ? 0 : (atan2(lab[2], lab[1]) * 180 / .pi + 360).truncatingRemainder(dividingBy: 360)
            return "oklch(\(n(lab[0] * 100, 3))% \(n(chroma, 5)) \(n(hue, 3))\(color.alpha < 1 ? " / \(a)" : ""))"
        case .displayP3:
            let p3 = color.nsColor.usingColorSpace(.displayP3)!
            return "color(display-p3 \(n(p3.redComponent)) \(n(p3.greenComponent)) \(n(p3.blueComponent))\(color.alpha < 1 ? " / \(a)" : ""))"
        case .swiftUI:
            let c = color.components
            return "Color(.sRGB, red: \(n(c[0])), green: \(n(c[1])), blue: \(n(c[2])), opacity: \(a))"
        case .appKit:
            let c = color.components
            return "NSColor(srgbRed: \(n(c[0])), green: \(n(c[1])), blue: \(n(c[2])), alpha: \(a))"
        case .cssVariable:
            let token = name.lowercased().unicodeScalars.map { CharacterSet.alphanumerics.contains($0) && $0.isASCII ? String($0) : "-" }.joined()
            return "--color-\(token.isEmpty ? "prismdrop" : token): \(color.alpha < 1 ? color.hexAlpha : color.hex);"
        case .tailwind: return rgbValue // Valid in Tailwind theme.colors and CSS @theme tokens.
        }
    }

struct ContrastResult {
    let foreground: SampledColor
    let background: SampledColor
    let ratio: Double
    init(foreground: SampledColor, background: SampledColor, canvas: SampledColor = .white) {
        self.foreground = foreground; self.background = background
        // Keep double precision through compositing: don't quantize translucent channels.
        let back = zip(background.components, canvas.components).map { $0 * background.alpha + $1 * (1 - background.alpha) }
        let front = zip(foreground.components, back).map { $0 * foreground.alpha + $1 * (1 - foreground.alpha) }
        func luminance(_ c: [Double]) -> Double {
            let l = c.map(SampledColor.linear)
            return 0.2126*l[0] + 0.7152*l[1] + 0.0722*l[2]
        }
        let l1 = luminance(front), l2 = luminance(back)
        ratio = (max(l1, l2) + 0.05) / (min(l1, l2) + 0.05)
    }
    var normalAA: Bool { ratio >= 4.5 }
    var normalAAA: Bool { ratio >= 7 }
    var largeAA: Bool { ratio >= 3 }
    var largeAAA: Bool { ratio >= 4.5 }

    // Exact minimum Euclidean distance in 8-bit encoded sRGB at the current
    // alpha. For each R/G pair, luminance monotonicity gives two allowed B
    // intervals; only the B nearest the source can minimize RGB distance.
    // This searches the whole gamut in 65,536 pairs, not 16 million trials.
    func suggestion(target: Double = 4.5, canvas: SampledColor = .white) -> SampledColor? {
        guard ratio < target else { return nil }
        let back = zip(background.components, canvas.components).map { $0 * background.alpha + $1 * (1 - background.alpha) }
        let backL = 0.2126 * SampledColor.linear(back[0]) + 0.7152 * SampledColor.linear(back[1]) + 0.0722 * SampledColor.linear(back[2])
        let darkLimit = (backL + 0.05) / target - 0.05
        let lightLimit = target * (backL + 0.05) - 0.05
        let sourceR = Int(foreground.red), sourceG = Int(foreground.green), sourceB = Int(foreground.blue)
        for alpha in foreground.alpha < 1 ? [foreground.alpha, 1] : [1] {
            let tables = zip(back, [0.2126, 0.7152, 0.0722]).map { channel, coefficient in
                (0...255).map { coefficient * SampledColor.linear(Double($0) / 255 * alpha + channel * (1 - alpha)) }
            }
            var best: SampledColor?, bestDistance = Int.max
            for r in 0...255 {
                if Task.isCancelled { return nil }
                let dr = (r - sourceR) * (r - sourceR)
                if dr > bestDistance { continue }
                for g in 0...255 {
                    let rgDistance = dr + (g - sourceG) * (g - sourceG)
                    if rgDistance > bestDistance { continue }
                    let base = tables[0][r] + tables[1][g]
                    for dark in [true, false] {
                        if dark && base + tables[2][0] > darkLimit { continue }
                        if !dark && base + tables[2][255] < lightLimit { continue }
                        var low = 0, high = 255
                        while low < high {
                            let mid = dark ? (low + high + 1) / 2 : (low + high) / 2
                            let luminance = base + tables[2][mid]
                            if dark {
                                if luminance <= darkLimit { low = mid } else { high = mid - 1 }
                            } else {
                                if luminance >= lightLimit { high = mid } else { low = mid + 1 }
                            }
                        }
                        let nearestB = dark ? min(sourceB, low) : max(sourceB, low)
                        // Adjacent B handles floating-point equality at a boundary;
                        // final eligibility always uses the same contrast evaluator.
                        for b in [nearestB, nearestB + (dark ? -1 : 1)] where (0...255).contains(b) {
                            let distance = rgDistance + (b - sourceB) * (b - sourceB)
                            guard distance < bestDistance else { continue }
                            let candidate = SampledColor(red: UInt8(r), green: UInt8(g), blue: UInt8(b), alpha: alpha)
                            if ContrastResult(foreground: candidate, background: background, canvas: canvas).ratio >= target {
                                best = candidate; bestDistance = distance
                            }
                        }
                    }
                }
            }
            if let best { return best }
        }
        return nil
    }
}
```

```swift
import Foundation

struct SavedColor: Codable, Identifiable, Equatable {
    var id = UUID()
    var color: SampledColor
    var name = ""
    var favorite = false
    var pickedAt = Date()
}

struct SavedPalette: Codable, Identifiable, Equatable {
    var id = UUID()
    var name: String
    var colors: [SampledColor]
    var createdAt = Date()
}

struct ColorLibrary: Codable, Equatable {
    var version = 1
    var history: [SavedColor] = []
    var palettes: [SavedPalette] = []
    static let recentLimit = 100

    mutating func record(_ color: SampledColor, at date: Date = Date()) {
        if let index = history.firstIndex(where: { $0.color.identity == color.identity }) {
            var entry = history.remove(at: index)
            entry.color = color
            entry.pickedAt = date
            history.insert(entry, at: 0)
        } else {
            history.insert(SavedColor(color: color, pickedAt: date), at: 0)
        }
        trim()
    }

    mutating func trim() {
        var recentCount = 0
        history = history.filter {
            if $0.favorite { return true }
            recentCount += 1
            return recentCount <= Self.recentLimit
        }
    }

    mutating func clearHistory() { history.removeAll { !$0.favorite } }

    func encoded() throws -> Data { try JSONEncoder().encode(self) }

    static func load(from url: URL) throws -> ColorLibrary {
        guard FileManager.default.fileExists(atPath: url.path) else { return ColorLibrary() }
        let data = try Data(contentsOf: url)
        let library = try JSONDecoder().decode(ColorLibrary.self, from: data)
        guard library.version == 1 else { throw LibraryError.unsupportedVersion }
        let colors = library.history.map(\.color) + library.palettes.flatMap(\.colors)
        guard colors.allSatisfy({ $0.alpha.isFinite && (0...1).contains($0.alpha) }),
              Set(library.history.map(\.id)).count == library.history.count,
              Set(library.history.map { $0.color.identity }).count == library.history.count,
              Set(library.palettes.map(\.id)).count == library.palettes.count else { throw LibraryError.invalidData }
        var result = library
        result.history.sort { $0.pickedAt > $1.pickedAt }
        result.trim()
        return result
    }

    static func write(_ data: Data, to url: URL) throws {
        try FileManager.default.createDirectory(at: url.deletingLastPathComponent(), withIntermediateDirectories: true)
        try data.write(to: url, options: .atomic)
    }

    enum LibraryError: LocalizedError {
        case unsupportedVersion, invalidData
        var errorDescription: String? {
            switch self {
            case .unsupportedVersion: return "This library was made by a newer PrismDrop version."
            case .invalidData: return "The color library contains invalid or duplicate entries."
            }
        }
    }
}
```

```swift
import Foundation
import CoreGraphics
import ImageIO

enum PaletteExtractor {
    enum ExtractionError: LocalizedError {
        case unreadable, noColors, tooLarge
        var errorDescription: String? {
            switch self {
            case .unreadable: return "This image couldn’t be decoded. Try a PNG, JPEG, or HEIC file."
            case .noColors: return "This image has no visible colors. Try an image with opaque content."
            case .tooLarge: return "Choose an image smaller than 100 MB."
            }
        }
    }

    static func extract(url: URL) throws -> [SampledColor] {
        let size = try url.resourceValues(forKeys: [.fileSizeKey]).fileSize ?? 0
        guard size <= 100 * 1024 * 1024 else { throw ExtractionError.tooLarge }
        guard let source = CGImageSourceCreateWithURL(url as CFURL, [kCGImageSourceShouldCache: false] as CFDictionary) else { throw ExtractionError.unreadable }
        return try extract(source: source)
    }

    static func extract(data: Data) throws -> [SampledColor] {
        guard data.count <= 100 * 1024 * 1024 else { throw ExtractionError.tooLarge }
        guard let source = CGImageSourceCreateWithData(data as CFData, [kCGImageSourceShouldCache: false] as CFDictionary) else { throw ExtractionError.unreadable }
        return try extract(source: source)
    }

    private static func extract(source: CGImageSource) throws -> [SampledColor] {
        try Task.checkCancellation()
        let options: [CFString: Any] = [kCGImageSourceCreateThumbnailFromImageAlways: true,
                                        kCGImageSourceCreateThumbnailWithTransform: true,
                                        kCGImageSourceThumbnailMaxPixelSize: 256,
                                        kCGImageSourceShouldCacheImmediately: true]
        guard let image = CGImageSourceCreateThumbnailAtIndex(source, 0, options as CFDictionary) else { throw ExtractionError.unreadable }
        return try quantize(image)
    }

    // Deterministic weighted OKLab k-means over a quantized histogram of every
    // thumbnail pixel. No random samples, and no full-resolution bitmap allocation.
    static func quantize(_ image: CGImage) throws -> [SampledColor] {
        guard image.width <= 256, image.height <= 256, image.width > 0, image.height > 0 else { throw ExtractionError.tooLarge }
        let count = image.width * image.height
        var pixels = [UInt8](repeating: 0, count: count * 4)
        guard let context = CGContext(data: &pixels, width: image.width, height: image.height, bitsPerComponent: 8,
                                      bytesPerRow: image.width * 4, space: CGColorSpace(name: CGColorSpace.sRGB)!,
                                      bitmapInfo: CGBitmapInfo.byteOrder32Big.rawValue | CGImageAlphaInfo.premultipliedLast.rawValue) else { throw ExtractionError.unreadable }
        context.draw(image, in: CGRect(x: 0, y: 0, width: image.width, height: image.height))
        struct Bin { var r = 0.0; var g = 0.0; var b = 0.0; var a = 0.0; var count = 0.0 }
        var histogram: [Int: Bin] = [:]
        for index in 0..<count {
            let i = index * 4, a = Double(pixels[i + 3]) / 255
            guard a >= 0.1 else { continue }
            let r = min(255, Double(pixels[i]) / a), g = min(255, Double(pixels[i + 1]) / a), b = min(255, Double(pixels[i + 2]) / a)
            let key = (Int(r) >> 3) << 13 | (Int(g) >> 3) << 8 | (Int(b) >> 3) << 3 | (Int(pixels[i + 3]) >> 5)
            var bin = histogram[key] ?? Bin()
            bin.r += r; bin.g += g; bin.b += b; bin.a += a; bin.count += 1
            histogram[key] = bin
        }
        struct Point { let lab: [Double]; let weight: Double; let alpha: Double }
        let points = histogram.sorted { $0.key < $1.key }.map { _, bin -> Point in
            let color = SampledColor.from([bin.r, bin.g, bin.b].map { $0 / bin.count / 255 }, alpha: bin.a / bin.count)
            return Point(lab: color.lab, weight: bin.count * color.alpha, alpha: color.alpha)
        }
        guard !points.isEmpty else { throw ExtractionError.noColors }
        func distance(_ a: [Double], _ b: [Double]) -> Double { zip(a, b).reduce(0) { $0 + pow($1.0 - $1.1, 2) } }
        var centers = [points.max { $0.weight < $1.weight }!.lab]
        while centers.count < min(8, points.count) {
            let candidate = points.max { a, b in
                let da = centers.map { distance(a.lab, $0) }.min()!
                let db = centers.map { distance(b.lab, $0) }.min()!
                return da * sqrt(a.weight) < db * sqrt(b.weight)
            }!
            guard centers.map({ distance(candidate.lab, $0) }).min()! > 0.000001 else { break }
            centers.append(candidate.lab)
        }
        var weights = [Double](repeating: 0, count: centers.count)
        var alphas = weights
        for _ in 0..<24 {
            try Task.checkCancellation()
            var sums = centers.map { _ in [Double](repeating: 0, count: 3) }
            weights = centers.map { _ in 0 }; alphas = weights
            for p in points {
                let i = centers.indices.min { distance(p.lab, centers[$0]) < distance(p.lab, centers[$1]) }!
                for channel in 0..<3 { sums[i][channel] += p.lab[channel] * p.weight }
                weights[i] += p.weight; alphas[i] += p.alpha * p.weight
            }
            var movement = 0.0
            for i in centers.indices where weights[i] > 0 {
                let next = sums[i].map { $0 / weights[i] }
                movement += distance(next, centers[i]); centers[i] = next
            }
            if movement < 0.0000001 { break }
        }
        var result: [SampledColor] = []
        for i in centers.indices.sorted(by: { weights[$0] > weights[$1] }) where weights[i] > 0 {
            let color = SampledColor.fromLab(centers[i], alpha: alphas[i] / weights[i])
            if !result.contains(where: { distance($0.lab, color.lab) < 0.055 * 0.055 }) { result.append(color) }
        }
        return result
    }
}
```

```swift
        path.addLine(to: CGPoint(x: x + 0.01 * s, y: y - 0.29 * s))
        path.addLine(to: CGPoint(x: x + 0.09 * s, y: y - 0.22 * s))
        path.closeSubpath()
        return path
    }
}

private enum OverlayAttachment {
    case below, above

    var revealAnchor: UnitPoint {
        self == .below ? .top : .bottom
    }
}

private struct CursorBubble: Shape {
    let attachment: OverlayAttachment
    let pointerX: CGFloat

    func path(in rect: CGRect) -> Path {
        let inset: CGFloat = 3
        let radius: CGFloat = 14
        let left = inset
        let right = rect.width - inset
        let tip = min(right - radius - 9, max(left + radius + 9, pointerX))
        let top: CGFloat = attachment == .below ? 9 : 1
        let bottom: CGFloat = attachment == .below ? rect.height - 1 : rect.height - 9
        var path = Path()
        path.move(to: CGPoint(x: left + radius, y: top))
        if attachment == .below {
            path.addLine(to: CGPoint(x: tip - 8, y: top))
            path.addLine(to: CGPoint(x: tip, y: 1))
            path.addLine(to: CGPoint(x: tip + 8, y: top))
        }
        path.addLine(to: CGPoint(x: right - radius, y: top))
        path.addQuadCurve(to: CGPoint(x: right, y: top + radius),
                          control: CGPoint(x: right, y: top))
        path.addLine(to: CGPoint(x: right, y: bottom - radius))
        path.addQuadCurve(to: CGPoint(x: right - radius, y: bottom),
                          control: CGPoint(x: right, y: bottom))
        if attachment == .above {
            path.addLine(to: CGPoint(x: tip + 8, y: bottom))
            path.addLine(to: CGPoint(x: tip, y: rect.height - 1))
            path.addLine(to: CGPoint(x: tip - 8, y: bottom))
        }
        path.addLine(to: CGPoint(x: left + radius, y: bottom))
        path.addQuadCurve(to: CGPoint(x: left, y: bottom - radius),
                          control: CGPoint(x: left, y: bottom))
        path.addLine(to: CGPoint(x: left, y: top + radius))
        path.addQuadCurve(to: CGPoint(x: left + radius, y: top),
                          control: CGPoint(x: left, y: top))
        path.closeSubpath()
        return path
    }
}

@MainActor
private final class OverlayPanel {
    let model = OverlayModel()
    private let panel: NSPanel
    private var presentationID = 0

    init() {
        panel = NSPanel(contentRect: CGRect(x: 0, y: 0, width: 186, height: 62),
                        styleMask: [.borderless, .nonactivatingPanel],
                        backing: .buffered, defer: false)
        panel.contentView = NSHostingView(rootView: OverlayView(model: model))
        panel.backgroundColor = .clear
        panel.isOpaque = false
        panel.hasShadow = false
        panel.level = .floating
        panel.collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary]
        panel.ignoresMouseEvents = true
        panel.hidesOnDeactivate = false
        panel.animationBehavior = .none
        panel.orderOut(nil)
    }

    func show(at mouse: CGPoint, shortcut: String) {
        presentationID += 1
        model.shortcut = shortcut
        model.color = nil
        model.message = nil
        model.copied = false
        position(at: mouse, snap: true)
        panel.orderFrontRegardless()
        model.isPresented = true
    }

    func update(_ sample: SampledColor, at mouse: CGPoint) {
        model.color = sample
        model.message = nil
        model.haloColor = Color(nsColor: sample.nsColor)
        position(at: mouse, snap: false)
    }

    func position(at mouse: CGPoint, snap: Bool = false) {
        guard panel.isVisible || snap else { return }
        let screen = NSScreen.screens.first { $0.frame.contains(mouse) } ?? NSScreen.main
        let safe = screen?.frame.insetBy(dx: 2, dy: 2) ?? CGRect(x: mouse.x - 400, y: mouse.y - 300, width: 800, height: 600)
        let width: CGFloat = 186
        let height: CGFloat = 62
        let gap: CGFloat = 2
        let below = mouse.y - gap - height >= safe.minY
        let x = min(max(mouse.x - width / 2, safe.minX), safe.maxX - width)
        let y = below ? mouse.y - gap - height : mouse.y + gap
        let attachment: OverlayAttachment = below ? .below : .above
        if model.attachment != attachment { model.attachment = attachment }
        let pointerX = mouse.x - x
        if model.pointerX != pointerX { model.pointerX = pointerX }
        panel.setFrameOrigin(CGPoint(x: x, y: y))
    }

    func showError(_ message: String) {
        model.color = nil
        model.message = message
    }

    func markCopied() {
        model.copied = true
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.7) { [weak self] in
            self?.model.copied = false
        }
    }

    func hide() {
        presentationID += 1
        let id = presentationID
        model.isPresented = false
        DispatchQueue.main.asyncAfter(deadline: .now() + (NSWorkspace.shared.accessibilityDisplayShouldReduceMotion ? 0.10 : 0.22)) { [weak self] in
            guard let self, self.presentationID == id else { return }
            self.panel.orderOut(nil)
            self.model.color = nil
        }
    }
}

@MainActor
private final class PixelSampler {
    private var displays: [CGDirectDisplayID: SCDisplay] = [:]
    private var isRefreshing = false

    func refreshDisplays() async throws {
        guard !isRefreshing else { return }
        isRefreshing = true
        defer { isRefreshing = false }
        let content = try await SCShareableContent.excludingDesktopWindows(false, onScreenWindowsOnly: false)
        displays = Dictionary(uniqueKeysWithValues: content.displays.map { ($0.displayID, $0) })
    }

    func sample(at point: CGPoint) async throws -> SampledColor {
        var displayID: CGDirectDisplayID = 0
        var count: UInt32 = 0
        let error = CGGetDisplaysWithPoint(point, 1, &displayID, &count)
        guard error == .success, count > 0 else { throw SamplingError.noDisplay }
        if displays[displayID] == nil { try await refreshDisplays() }
        guard let display = displays[displayID] else { throw SamplingError.noDisplay }

        let bounds = CGDisplayBounds(displayID)
        let localX = point.x - bounds.minX
        let localY = point.y - bounds.minY
        guard localX >= 0, localY >= 0, localX < bounds.width, localY < bounds.height else {
            throw SamplingError.noDisplay
        }

        let screen = NSScreen.screens.first {
            ($0.deviceDescription[NSDeviceDescriptionKey("NSScreenNumber")] as? NSNumber)?.uint32Value == displayID
        }
        let scale = screen?.backingScaleFactor ?? 1
        let cellX = floor(localX)
        let cellY = floor(localY)
        let width = max(1, Int(ceil(scale)))
        let config = SCStreamConfiguration()
        config.sourceRect = CGRect(x: cellX, y: cellY, width: 1, height: 1)
        config.width = width
        config.height = width
        config.scalesToFit = false
        config.showsCursor = false
        config.pixelFormat = kCVPixelFormatType_32BGRA
        config.colorSpaceName = CGColorSpace.sRGB as CFString
        config.captureResolution = .best
        let filter = SCContentFilter(display: display, excludingApplications: [], exceptingWindows: [])
        let image = try await SCScreenshotManager.captureImage(contentFilter: filter, configuration: config)
        guard image.width > 0, image.height > 0 else { throw SamplingError.emptyImage }
        let pixelX = min(image.width - 1, max(0, Int((localX - cellX) * CGFloat(image.width))))
        let pixelY = min(image.height - 1, max(0, Int((localY - cellY) * CGFloat(image.height))))
        guard let pixel = image.cropping(to: CGRect(x: pixelX, y: pixelY, width: 1, height: 1)) else {
            throw SamplingError.emptyImage
        }
        var bytes = [UInt8](repeating: 0, count: 4)
        let colorSpace = CGColorSpace(name: CGColorSpace.sRGB)!
        let bitmapInfo = CGBitmapInfo.byteOrder32Big.rawValue | CGImageAlphaInfo.premultipliedLast.rawValue
        guard let context = CGContext(data: &bytes, width: 1, height: 1,
                                      bitsPerComponent: 8, bytesPerRow: 4,
                                      space: colorSpace, bitmapInfo: bitmapInfo) else {
            throw SamplingError.emptyImage
        }
        context.interpolationQuality = .none
        context.setBlendMode(.copy)
        context.draw(pixel, in: CGRect(x: 0, y: 0, width: 1, height: 1))
        return SampledColor(red: bytes[0], green: bytes[1], blue: bytes[2])
    }

    enum SamplingError: LocalizedError {
        case noDisplay, emptyImage
        var errorDescription: String? {
            switch self {
            case .noDisplay: return "No capture display contains the pointer."
            case .emptyImage: return "ScreenCaptureKit returned no pixel."
            }
        }
    }
}

@MainActor
private final class AppController: NSObject, NSApplicationDelegate, NSMenuDelegate {
    private let overlay = OverlayPanel()
    private let sampler = PixelSampler()
    private let studio: StudioStore
    private var studioWindow: StudioWindow?
    private var sessionColor: SampledColor?
    private var statusItem: NSStatusItem!
    private var menu: NSMenu!
    private var timer: Timer?
    private var eventTap: CFMachPort?
    private var eventTapSource: CFRunLoopSource?
    private var isHolding = false
    private var captureInFlight = false
    private var generation = 0
    private var lastColor: SampledColor?
    private var lastCaptureError: String?
    private var lastFailureAt = Date.distantPast
    private var lastPermissionCheck = Date.distantPast
    private var inputAllowed = false
    private var screenAllowed = false
    private var pair = ModifierPair(rawValue: UserDefaults.standard.integer(forKey: "hotkeyPair")) ?? .optionShift

    override init() { studio = StudioStore(); super.init() }
    init(studio: StudioStore) { self.studio = studio; super.init() }

    func applicationDidFinishLaunching(_ notification: Notification) {
        installApplicationMenu()
        studio.onPickRequested = { [weak self] in self?.studioWindow?.window.orderOut(nil) }
        studio.onPickFinished = { [weak self] in
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.24) { self?.openStudio(tab: .contrast) }
        }
        studio.onFeaturesChanged = { [weak self] in
            guard let self else { return }
            if !self.studio.isEnabled(.picker), self.isHolding { self.endSampling() }
            self.overlay.model.copyFormat = self.studio.activeFormat
        }
        statusItem = NSStatusBar.system.statusItem(withLength: NSStatusItem.squareLength)
        if let image = NSImage(systemSymbolName: "eyedropper", accessibilityDescription: "PrismDrop") {
            image.isTemplate = true
            statusItem.button?.image = image
        } else {
            statusItem.button?.title = "◉"
        }
        statusItem.button?.toolTip = "PrismDrop — hold \(pair.title)"
        menu = NSMenu()
        menu.delegate = self
        statusItem.menu = menu
        rebuildMenu()

        // A listen-only event tap sees modifier-only flagsChanged events systemwide.
        // Polling the current state provides a release fail-safe if the tap is disabled.
        inputAllowed = CGPreflightListenEventAccess()
        screenAllowed = CGPreflightScreenCaptureAccess()
        installEventTapIfAllowed()
        NSLog("PrismDrop startup: screen=%@ input=%@ tap=%@",
              screenAllowed ? "allowed" : "denied",
              inputAllowed ? "allowed" : "denied",
              eventTap == nil ? "unavailable" : "active")
        timer = Timer(timeInterval: 1.0 / 60.0, repeats: true) { [weak self] _ in
            Task { @MainActor [weak self] in self?.tick() }
        }
        RunLoop.main.add(timer!, forMode: .common)

        if screenAllowed {
            Task { try? await sampler.refreshDisplays() }
        }
    }

    func applicationWillTerminate(_ notification: Notification) {
        studio.cancelExtraction()
        studio.flush()
        timer?.invalidate()
        if let source = eventTapSource { CFRunLoopRemoveSource(CFRunLoopGetMain(), source, .commonModes) }
        if let tap = eventTap { CFMachPortInvalidate(tap) }
    }

    private func installApplicationMenu() {
        let main = NSMenu()
        let app = NSMenuItem(); let appMenu = NSMenu(title: "PrismDrop")
        appMenu.addItem(item("Open Color Studio…", #selector(openStudioAction)))
        appMenu.addItem(.separator())
        let quitItem = item("Quit PrismDrop", #selector(quit)); quitItem.keyEquivalent = "q"
        appMenu.addItem(quitItem); app.submenu = appMenu; main.addItem(app)
        let edit = NSMenuItem(); let editMenu = NSMenu(title: "Edit")
        for (title, action, key) in [("Undo", "undo:", "z"), ("Redo", "redo:", "Z"),
                                     ("Cut", "cut:", "x"), ("Copy", "copy:", "c"),
                                     ("Paste", "paste:", "v"), ("Select All", "selectAll:", "a")] {
            editMenu.addItem(NSMenuItem(title: title, action: Selector(action), keyEquivalent: key))
        }
        edit.submenu = editMenu; main.addItem(edit)
        let windowItem = NSMenuItem(); let windowMenu = NSMenu(title: "Window")
        windowMenu.addItem(NSMenuItem(title: "Close", action: #selector(NSWindow.performClose(_:)), keyEquivalent: "w"))
        windowItem.submenu = windowMenu; main.addItem(windowItem)
        NSApp.mainMenu = main
    }

    private func installEventTapIfAllowed() {
        guard inputAllowed, eventTap == nil else { return }
        let mask = (CGEventMask(1) << CGEventType.flagsChanged.rawValue) |
                   (CGEventMask(1) << CGEventType.keyDown.rawValue)
        let refcon = UnsafeMutableRawPointer(Unmanaged.passUnretained(self).toOpaque())
        eventTap = CGEvent.tapCreate(tap: .cgSessionEventTap, place: .headInsertEventTap,
                                     options: .listenOnly, eventsOfInterest: mask,
                                     callback: { _, type, event, refcon in
            guard let refcon else { return Unmanaged.passUnretained(event) }
            let controller = Unmanaged<AppController>.fromOpaque(refcon).takeUnretainedValue()
            DispatchQueue.main.async { controller.handleEvent(type: type, event: event) }
            return Unmanaged.passUnretained(event)
        }, userInfo: refcon)
        guard let tap = eventTap else {
            lastCaptureError = "Input Monitoring was allowed, but the event tap could not start. Relaunch PrismDrop."
            return
        }
        let source = CFMachPortCreateRunLoopSource(kCFAllocatorDefault, tap, 0)
        eventTapSource = source
        CFRunLoopAddSource(CFRunLoopGetMain(), source, .commonModes)
        CGEvent.tapEnable(tap: tap, enable: true)
    }

    private func handleEvent(type: CGEventType, event: CGEvent) {
        if type == .tapDisabledByTimeout || type == .tapDisabledByUserInput {
            if let tap = eventTap { CGEvent.tapEnable(tap: tap, enable: true) }
            return
        }
        if type == .flagsChanged {
            updateHeldState(flags: event.flags)
        } else if type == .keyDown,
                  event.getIntegerValueField(.keyboardEventKeycode) == 8,
                  pair.matches(event.flags), isHolding {
            copyLastColor()
        } else if type == .keyDown, isHolding, studio.isEnabled(.formats), pair.matches(event.flags),
                  [123, 124].contains(event.getIntegerValueField(.keyboardEventKeycode)) {
            let choices = CopyFormat.allCases
            let index = choices.firstIndex(of: studio.defaultFormat) ?? 0
            let direction = event.getIntegerValueField(.keyboardEventKeycode) == 124 ? 1 : -1
            studio.defaultFormat = choices[(index + direction + choices.count) % choices.count]
            overlay.model.copyFormat = studio.activeFormat
        }
    }

    private func tick() {
        if Date().timeIntervalSince(lastPermissionCheck) >= 1 {
            lastPermissionCheck = Date()
            let allowed = CGPreflightListenEventAccess()
            if allowed != inputAllowed {
                NSLog("PrismDrop Input Monitoring changed: %@", allowed ? "allowed" : "denied")
            }
            inputAllowed = allowed
            if inputAllowed { installEventTapIfAllowed() }
        }
        guard inputAllowed else {
            if isHolding { endSampling() }
            return
        }
        let flags = CGEventSource.flagsState(.combinedSessionState)
        updateHeldState(flags: flags)
        if isHolding {
            overlay.position(at: NSEvent.mouseLocation)
            requestSample()
        }
    }

    private func updateHeldState(flags: CGEventFlags) {
        let shouldHold = studio.isEnabled(.picker) && pair.matches(flags)
        guard shouldHold != isHolding else { return }
        if shouldHold { beginSampling() } else { endSampling() }
    }

    private func beginSampling() {
        isHolding = true
        sessionColor = nil
        overlay.model.copyFormat = studio.activeFormat
        generation += 1
        overlay.show(at: NSEvent.mouseLocation, shortcut: pair.shortTitle)
        screenAllowed = CGPreflightScreenCaptureAccess()
        if !screenAllowed {
            overlay.showError("SCREEN ACCESS NEEDED")
            return
        }
        requestSample() // Launch first capture before any reveal animation finishes.
    }

    private func endSampling() {
        isHolding = false
        generation += 1 // Invalidate any capture that finishes after key release.
        overlay.hide()
        if let color = sessionColor { studio.acceptPicked(color, finished: true) }
        sessionColor = nil
    }

    private func requestSample() {
        guard isHolding, !captureInFlight, screenAllowed else { return }
        guard let location = CGEvent(source: nil)?.location else { return }
        captureInFlight = true
        let requestGeneration = generation
        Task { @MainActor in
            defer { captureInFlight = false }
            do {
                let sample = try await sampler.sample(at: location)
                guard isHolding, generation == requestGeneration else { return }
                lastColor = sample
                sessionColor = sample
                lastCaptureError = nil
                overlay.update(sample, at: NSEvent.mouseLocation)
            } catch {
                guard isHolding, generation == requestGeneration else { return }
                if Date().timeIntervalSince(lastFailureAt) > 2 {
                    lastCaptureError = error.localizedDescription
                    lastFailureAt = Date()
                    overlay.showError("CAPTURE UNAVAILABLE")
                    Task { try? await sampler.refreshDisplays() }
                }
            }
        }
    }

    private func copyLastColor() {
        guard let color = isHolding ? sessionColor : lastColor else { return }
        if studio.copy(color) {
            studio.acceptPicked(color, finished: false)
            overlay.markCopied()
        }
    }

    private func openStudio(tab: StudioTab? = nil) {
        if let tab { studio.tab = tab }
        if studioWindow == nil { studioWindow = StudioWindow(store: studio, shortcut: pair.title) }
        studioWindow?.show()
    }

    @objc private func openStudioAction() { openStudio() }
    func openStudioActionForLaunch() { openStudio() }
    @objc private func selectFormat(_ sender: NSMenuItem) {
        guard CopyFormat.allCases.indices.contains(sender.tag) else { return }
        studio.defaultFormat = CopyFormat.allCases[sender.tag]
        overlay.model.copyFormat = studio.activeFormat
    }
    @objc private func toggleFeature(_ sender: NSMenuItem) {
        guard StudioFeature.allCases.indices.contains(sender.tag) else { return }
        let feature = StudioFeature.allCases[sender.tag]
        studio.setEnabled(feature, !studio.isEnabled(feature))
    }

    func menuNeedsUpdate(_ menu: NSMenu) { rebuildMenu() }

    private func rebuildMenu() {
        menu.removeAllItems()
        let title = NSMenuItem(title: "PrismDrop", action: nil, keyEquivalent: "")
        title.isEnabled = false
        menu.addItem(title)
        let instruction = NSMenuItem(title: "Hold \(pair.title) to sample", action: nil, keyEquivalent: "")
        instruction.isEnabled = false
        menu.addItem(instruction)
        menu.addItem(.separator())

        menu.addItem(item("Open Color Studio…", #selector(openStudioAction)))
        if let role = studio.pendingPick {
            let picking = NSMenuItem(title: "Release the picker to set \(role.rawValue.lowercased())", action: nil, keyEquivalent: "")
            picking.isEnabled = false; menu.addItem(picking)
        }
        menu.addItem(.separator())
        let screenGranted = CGPreflightScreenCaptureAccess()
        let inputGranted = CGPreflightListenEventAccess()
        menu.addItem(item("Screen Recording: \(screenGranted ? "Allowed" : "Needs permission")", #selector(requestScreenAccess)))
        menu.addItem(item("Input Monitoring: \(inputGranted ? "Allowed" : "Needs permission")", #selector(requestInputAccess)))
        if !inputGranted || !screenGranted {
            menu.addItem(item("Open Privacy Settings…", #selector(openPrivacySettings)))
        }
        if let error = lastCaptureError {
            let errorItem = NSMenuItem(title: "Issue: \(String(error.prefix(80)))", action: nil, keyEquivalent: "")
            errorItem.isEnabled = false
            menu.addItem(errorItem)
        }
        menu.addItem(.separator())

        let copy = item(lastColor.map { "Copy Last \(studio.activeFormat.title)  \($0.hex)" } ?? "Copy Last Color (none yet)", #selector(copyMenuAction))
        copy.isEnabled = lastColor != nil
        menu.addItem(copy)
        if studio.isEnabled(.formats) {
            let formatMenu = NSMenuItem(title: "Copy Format: \(studio.defaultFormat.title)", action: nil, keyEquivalent: "")
            let formats = NSMenu()
            for (index, format) in CopyFormat.allCases.enumerated() {
                let choice = item(format.title, #selector(selectFormat(_:)))
                choice.tag = index; choice.state = format == studio.defaultFormat ? .on : .off
                formats.addItem(choice)
            }
            formatMenu.submenu = formats; menu.addItem(formatMenu)
        }
        let featuresMenu = NSMenuItem(title: "Features", action: nil, keyEquivalent: "")
        let features = NSMenu()
        for (index, feature) in StudioFeature.allCases.enumerated() {
            let choice = item(feature.title, #selector(toggleFeature(_:)))
            choice.tag = index; choice.state = studio.isEnabled(feature) ? .on : .off
            features.addItem(choice)
        }
        featuresMenu.submenu = features; menu.addItem(featuresMenu)
        let hotkey = NSMenuItem(title: "Activation hotkey", action: nil, keyEquivalent: "")
        let sub = NSMenu()
        for value in ModifierPair.allCases {
            let choice = item(value.title, #selector(selectHotkey(_:)))
            choice.tag = value.rawValue
            choice.state = value == pair ? .on : .off
            sub.addItem(choice)
        }
        hotkey.submenu = sub
        menu.addItem(hotkey)
        menu.addItem(.separator())
        menu.addItem(item("Quit PrismDrop", #selector(quit)))
    }

    private func item(_ title: String, _ action: Selector) -> NSMenuItem {
```

```swift
enum ContrastRole: String { case foreground = "Foreground", background = "Background" }

@MainActor
final class StudioStore: ObservableObject {
    @Published private(set) var library = ColorLibrary()
    @Published private(set) var enabled: Set<StudioFeature>
    @Published var tab: StudioTab = .history
    @Published var selected = SampledColor(red: 94, green: 177, blue: 242)
    @Published var defaultFormat: CopyFormat { didSet { defaults.set(defaultFormat.rawValue, forKey: "defaultCopyFormat") } }
    @Published var foreground = SampledColor(red: 89, green: 99, blue: 117)
    @Published var background = SampledColor.white
    @Published var canvas = SampledColor.white
    @Published var pendingPick: ContrastRole?
    @Published private(set) var extracted: [SampledColor] = []
    @Published private(set) var extractionName = "Image palette"
    @Published private(set) var isExtracting = false
    @Published var errorMessage: String?
    @Published private(set) var persistenceError: String?
    @Published private(set) var copiedText: String?
    var onPickRequested: (() -> Void)?
    var onPickFinished: (() -> Void)?
    var onFeaturesChanged: (() -> Void)?
    let libraryURL: URL
    private let defaults: UserDefaults
    private let writeQueue = DispatchQueue(label: "local.prismdrop.library", qos: .utility)
    private var revision = 0
    private var storageBlocked = false
    private var extractionID = 0
    private var worker: Task<[SampledColor], Error>?
    private var copiedID = 0

    init(defaults: UserDefaults = .standard, libraryURL: URL? = nil) {
        self.defaults = defaults
        self.libraryURL = libraryURL ?? FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("PrismDrop", isDirectory: true).appendingPathComponent("library.json")
        enabled = Set(StudioFeature.allCases.filter { defaults.object(forKey: "feature.\($0.rawValue)") as? Bool ?? true })
        defaultFormat = CopyFormat(rawValue: defaults.string(forKey: "defaultCopyFormat") ?? "hex") ?? .hex
        reload()
    }

    func isEnabled(_ feature: StudioFeature) -> Bool { enabled.contains(feature) }
    var activeFormat: CopyFormat { isEnabled(.formats) ? defaultFormat : .hex }

    func setEnabled(_ feature: StudioFeature, _ value: Bool) {
        if value { enabled.insert(feature) } else { enabled.remove(feature) }
        defaults.set(value, forKey: "feature.\(feature.rawValue)")
        if !value && feature == .palettes { cancelExtraction() }
        if !value && (feature == .contrast || feature == .picker) { pendingPick = nil }
        onFeaturesChanged?()
    }

    func reload() {
        writeQueue.sync {} // Drain old writes before reading or retrying.
        do {
            library = try ColorLibrary.load(from: libraryURL)
            storageBlocked = false; persistenceError = nil
        } catch {
            storageBlocked = true
            persistenceError = "Couldn’t load the color library: \(error.localizedDescription) The original file is unchanged. Restore it, then Retry."
        }
    }

    func persist() {
        guard !storageBlocked else { return }
        revision += 1
        let current = revision, url = libraryURL
        do {
            let data = try library.encoded()
            writeQueue.async { [weak self] in
                do {
                    try ColorLibrary.write(data, to: url)
                    DispatchQueue.main.async { [weak self] in
                        guard let self, self.revision == current else { return }
                        self.persistenceError = nil
                    }
                } catch {
                    let message = "Couldn’t save colors: \(error.localizedDescription) Your changes remain in memory. Retry saving before quitting."
                    DispatchQueue.main.async { [weak self] in
                        guard let self, self.revision == current else { return }
                        self.persistenceError = message
                    }
                }
            }
        } catch { persistenceError = error.localizedDescription }
    }

    func retryPersistence() { if storageBlocked { reload() } else { persist() } }
    func flush() { writeQueue.sync {} }

    func acceptPicked(_ color: SampledColor, finished: Bool) {
        selected = color
        if isEnabled(.history) { library.record(color); persist() }
        if let role = pendingPick, isEnabled(.contrast) {
            useForContrast(color, role: role)
            if finished { pendingPick = nil; onPickFinished?() }
        }
    }

    @discardableResult func copy(_ color: SampledColor, format: CopyFormat? = nil, name: String = "prismdrop") -> Bool {
        selected = color
        let value = (format ?? activeFormat).string(for: color, name: name)
        NSPasteboard.general.clearContents()
        guard NSPasteboard.general.setString(value, forType: .string) else {
            errorMessage = "Couldn’t write to the clipboard. Try copying again."; return false
        }
        copiedID += 1; let id = copiedID
        copiedText = value
        DispatchQueue.main.asyncAfter(deadline: .now() + 2) { [weak self] in
            guard self?.copiedID == id else { return }; self?.copiedText = nil
        }
        return true
    }

    func renameColor(_ id: UUID, _ name: String) {
        guard let i = library.history.firstIndex(where: { $0.id == id }) else { return }
        library.history[i].name = String(name.prefix(80)); persist()
    }
    func toggleFavorite(_ id: UUID) {
        guard let i = library.history.firstIndex(where: { $0.id == id }) else { return }
        library.history[i].favorite.toggle(); library.trim(); persist()
    }
    func favoriteSelected() {
        guard isEnabled(.history) else { return }
        library.record(selected)
        if let i = library.history.firstIndex(where: { $0.color.identity == selected.identity }) { library.history[i].favorite = true }
        persist()
    }
    func deleteColor(_ id: UUID) { library.history.removeAll { $0.id == id }; persist() }
    func clearHistory() { library.clearHistory(); persist() }

    func useForContrast(_ color: SampledColor, role: ContrastRole) {
        if role == .foreground { foreground = color } else { background = color }
    }
    func requestPick(_ role: ContrastRole) {
        guard isEnabled(.picker), isEnabled(.contrast) else { return }
        pendingPick = role; onPickRequested?()
    }
    func swapContrast() { let previous = foreground; foreground = background; background = previous }

    func importImage() {
        guard isEnabled(.palettes) else { return }
        let panel = NSOpenPanel()
        panel.allowedContentTypes = [.png, .jpeg, .heic]
        panel.canChooseDirectories = false; panel.allowsMultipleSelection = false
        panel.message = "Extract a palette from a PNG, JPEG, or HEIC image."
        panel.begin { [weak self] response in
            guard response == .OK, let url = panel.url else { return }
            Task { @MainActor in self?.extract(url: url) }
        }
    }

    func extract(url: URL) {
        startExtraction(name: url.deletingPathExtension().lastPathComponent) {
            let access = url.startAccessingSecurityScopedResource()
            defer { if access { url.stopAccessingSecurityScopedResource() } }
            return try PaletteExtractor.extract(url: url)
        }
    }
    func extract(data: Data, name: String = "Pasted image") {
        startExtraction(name: name) { try PaletteExtractor.extract(data: data) }
    }
    private func startExtraction(name: String, operation: @escaping @Sendable () throws -> [SampledColor]) {
        guard isEnabled(.palettes) else { return }
        cancelExtraction(); isExtracting = true; errorMessage = nil
        let id = extractionID
        let task = Task.detached(priority: .userInitiated, operation: operation)
        worker = task
        Task { [weak self] in
            do {
                let colors = try await task.value
                guard let self, self.extractionID == id, self.isEnabled(.palettes) else { return }
                self.extracted = colors; self.extractionName = String(name.prefix(80)); self.isExtracting = false; self.worker = nil
            } catch {
                guard let self, self.extractionID == id else { return }
                self.isExtracting = false; self.worker = nil
                if !(error is CancellationError) { self.errorMessage = error.localizedDescription }
            }
        }
    }
    func cancelExtraction() { extractionID += 1; worker?.cancel(); worker = nil; isExtracting = false }

    func pasteImage() {
        let board = NSPasteboard.general
        if let urls = board.readObjects(forClasses: [NSURL.self], options: [.urlReadingFileURLsOnly: true]) as? [URL], let url = urls.first {
            extract(url: url); return
        }
        for type in board.types ?? [] {
            if let uti = UTType(type.rawValue), uti.conforms(to: .image), let data = board.data(forType: type) {
                extract(data: data); return
            }
        }
        errorMessage = "The clipboard doesn’t contain an image. Copy an image, then try Paste Image again."
    }

    func handleDrop(_ providers: [NSItemProvider]) -> Bool {
        guard isEnabled(.palettes), let provider = providers.first else { return false }
        if provider.hasItemConformingToTypeIdentifier(UTType.fileURL.identifier) {
            cancelExtraction(); isExtracting = true
            let id = extractionID
            provider.loadDataRepresentation(forTypeIdentifier: UTType.fileURL.identifier) { [weak self] data, error in
                Task { @MainActor in
                    guard let self, self.extractionID == id, self.isEnabled(.palettes) else { return }
                    if let data, let url = URL(dataRepresentation: data, relativeTo: nil), url.isFileURL { self.extract(url: url) }
                    else { self.isExtracting = false; self.errorMessage = error?.localizedDescription ?? "The dropped file couldn’t be opened." }
                }
            }
            return true
        }
        guard let type = provider.registeredTypeIdentifiers.first(where: { UTType($0)?.conforms(to: .image) == true }) else { return false }
        cancelExtraction(); isExtracting = true
        let id = extractionID
        provider.loadDataRepresentation(forTypeIdentifier: type) { [weak self] data, error in
            Task { @MainActor in
                guard let self, self.extractionID == id, self.isEnabled(.palettes) else { return }
                if let data { self.extract(data: data, name: "Dropped image") }
                else { self.isExtracting = false; self.errorMessage = error?.localizedDescription ?? "The dropped image couldn’t be read." }
            }
        }
        return true
    }

    func saveExtracted(name: String) {
        guard !extracted.isEmpty, !isExtracting, isEnabled(.palettes) else { return }
        let title = name.trimmingCharacters(in: .whitespacesAndNewlines)
        library.palettes.insert(SavedPalette(name: String((title.isEmpty ? extractionName : title).prefix(80)), colors: extracted), at: 0)
        extracted = []; persist()
    }
    func renamePalette(_ id: UUID, _ name: String) {
        guard let i = library.palettes.firstIndex(where: { $0.id == id }) else { return }
        library.palettes[i].name = String(name.prefix(80)); persist()
    }
    func removePaletteColor(_ id: UUID, at index: Int) {
        guard let i = library.palettes.firstIndex(where: { $0.id == id }), library.palettes[i].colors.indices.contains(index) else { return }
        library.palettes[i].colors.remove(at: index); persist()
    }
    func deletePalette(_ id: UUID) { library.palettes.removeAll { $0.id == id }; persist() }
}
```

```swift
private func expect(_ condition: @autoclosure () throws -> Bool, _ message: String, file: StaticString = #filePath, line: UInt = #line) {
    checks += 1
    let passed: Bool

private func near(_ a: Double, _ b: Double, _ tolerance: Double = 0.0001) -> Bool { abs(a - b) < tolerance }
private func image(_ colors: [SampledColor], stripe: Int = 20) -> CGImage {
    let width = colors.count * stripe, height = 32
    var bytes = [UInt8]()
    for _ in 0..<height { for x in 0..<width {
        let c = colors[x / stripe]
        bytes += [UInt8((Double(c.red) * c.alpha).rounded()), UInt8((Double(c.green) * c.alpha).rounded()), UInt8((Double(c.blue) * c.alpha).rounded()), UInt8((255 * c.alpha).rounded())]
    } }
    return CGImage(width: width, height: height, bitsPerComponent: 8, bitsPerPixel: 32, bytesPerRow: width * 4,
                   space: CGColorSpace(name: CGColorSpace.sRGB)!, bitmapInfo: CGBitmapInfo(rawValue: CGImageAlphaInfo.premultipliedLast.rawValue),
                   provider: CGDataProvider(data: Data(bytes) as CFData)!, decode: nil, shouldInterpolate: false, intent: .defaultIntent)!
}
private func encoded(_ image: CGImage, type: UTType) -> Data? {
    let data = NSMutableData()
    guard let destination = CGImageDestinationCreateWithData(data, type.identifier as CFString, 1, nil) else { return nil }
    CGImageDestinationAddImage(destination, image, nil)
    return CGImageDestinationFinalize(destination) ? data as Data : nil
```

## `build.sh`

```sh
#!/bin/zsh
set -euo pipefail
project_dir=${0:A:h}
build_dir="$project_dir/build"
app_dir="$build_dir/PrismDrop.app"
mkdir -p "$app_dir/Contents/MacOS" "$app_dir/Contents/Resources"
for architecture in x86_64 arm64; do
  swiftc -parse-as-library -O -target "${architecture}-apple-macosx14.0" \
    -framework AppKit -framework SwiftUI -framework ScreenCaptureKit \
    "$project_dir"/Source/*.swift \
    -o "$build_dir/PrismDrop-$architecture"
done
lipo -create "$build_dir/PrismDrop-x86_64" "$build_dir/PrismDrop-arm64" \
  -output "$app_dir/Contents/MacOS/PrismDrop"
cp "$project_dir/Info.plist" "$app_dir/Contents/Info.plist"
cp "$project_dir/Assets/PrismDrop.icns" "$app_dir/Contents/Resources/PrismDrop.icns"
# Ad-hoc sign the local app. Rebuilds can change the identity used by privacy grants.
codesign --force --deep --sign - --identifier local.prismdrop.app "$app_dir"
echo "Built $app_dir"
```

## `install.sh`

```sh
#!/bin/zsh
set -euo pipefail
project_dir=${0:A:h}
built_app="$project_dir/build/PrismDrop.app"
source_changed=false
for source_file in "$project_dir"/Source/*.swift; do
  if [[ "$source_file" -nt "$built_app/Contents/MacOS/PrismDrop" ]]; then source_changed=true; fi
done
if [[ "$source_changed" == true || ! -x "$built_app/Contents/MacOS/PrismDrop" || \
      ! -f "$built_app/Contents/Resources/PrismDrop.icns" || \
      "$project_dir/Info.plist" -nt "$built_app/Contents/Info.plist" || \
      "$project_dir/Assets/PrismDrop.icns" -nt "$built_app/Contents/Resources/PrismDrop.icns" ]]; then
  "$project_dir/build.sh"
else
  echo "Using existing build (preserves its privacy identity)"
fi
mkdir -p "$HOME/Applications"
if [[ -d "$HOME/Applications/PrismDrop.app" ]] && \
   cmp -s "$built_app/Contents/MacOS/PrismDrop" "$HOME/Applications/PrismDrop.app/Contents/MacOS/PrismDrop" && \
   cmp -s "$built_app/Contents/Info.plist" "$HOME/Applications/PrismDrop.app/Contents/Info.plist" && \
   cmp -s "$built_app/Contents/Resources/PrismDrop.icns" "$HOME/Applications/PrismDrop.app/Contents/Resources/PrismDrop.icns"; then
  open "$HOME/Applications/PrismDrop.app"
  echo "Already installed: $HOME/Applications/PrismDrop.app"
  exit 0
fi
if [[ -d "$HOME/Applications/PrismDrop.app" ]]; then
  pkill -x PrismDrop >/dev/null 2>&1 || true
  rm -rf "$HOME/Applications/PrismDrop.app"
fi
ditto "$built_app" "$HOME/Applications/PrismDrop.app"
open "$HOME/Applications/PrismDrop.app"
echo "Installed and opened $HOME/Applications/PrismDrop.app"
```
